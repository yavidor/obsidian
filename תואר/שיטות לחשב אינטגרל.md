# אינטגרלים מידיים
$$
\begin{gather}
\int \cos x~dx=\sin x \\
\int \sin x~dx=-\cos x \\
\int \frac{1}{\cos ^{2}x}~dx=\int \sec ^{2}x~dx=\tan x \\
\int x^{a}~dx=\begin{cases}
\frac{x^{a+1}}{a+1} & a\neq -1 \\
\log \left| x \right|  & a=-1
\end{cases} \\
\int e^{ x }~dx=e^{ x } \\
\int \frac{1}{1+x^{2}}~dx=\arctan x=\tan ^{-1}x \\
\int \frac{1}{\sqrt{1-x^{2}}}~dx=\arcsin x=\sin ^{-1}x
\end{gather}
$$
