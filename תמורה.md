# הגדרה
[[פונקציה]] 
```dataviewjs
// Get all pages
const pages = dv.pages()

// This regex will find the contents of the callout [!quote]
// If you want a different type to be found, update the "quote" to the corresponding callout format
const regex = /\[\!(?<type>.*)\]\s?(?<title> .*)?\n\s*>\s*(?<content>.*)/m

const rows = []
for (const page of pages) {
  // Read the file contents
  const file = app.vault.getAbstractFileByPath(page.file.path)
  const contents = await app.vault.read(file);
  
  // Find quotes
  const quotes = contents.match(regex) || [];
  
  // Add all quotes to the rows
  if (quotes.length > 0) {
    rows.push([page.file.link, quotes.groups.type, quotes.groups.title, quotes.groups.content])
  }
}

// Create table
dv.table(['Link', 'Type', 'Title', 'Content'], rows)
```
